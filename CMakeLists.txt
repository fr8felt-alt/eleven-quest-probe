cmake_minimum_required(VERSION 3.20)
project(eleven_quest_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(eleven_quest_probe SHARED src/main.cpp)
target_include_directories(eleven_quest_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/external/scotland2/include")
target_link_libraries(eleven_quest_probe PRIVATE log)
set_target_properties(eleven_quest_probe PROPERTIES OUTPUT_NAME "eleven_quest_probe")
